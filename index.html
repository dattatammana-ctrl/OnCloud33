/** * OnCloud33 shared data store + activity log + direct group-request tabs * ------------------------------------------------------------- * Deploy this bound to the Google Sheet: * https://docs.google.com/spreadsheets/d/18OeRV5ePMSjf0hXQu_EeyhE3tbZS_x3k6BlU9MxPBKM/edit * * Setup: * 1. Open the Sheet → Extensions → Apps Script. * 2. Delete any existing code in Code.gs and paste this whole file. * 3. Save. * 4. Deploy → New deployment → type: "Web app". * - Execute as: Me * - Who has access: Anyone * 5. Click Deploy, authorize when prompted, copy the /exec URL. * 6. Paste that URL into whichever index.html version you're using as * STORAGE_WEBAPP_URL. If you create a NEW deployment later, you'll get * a NEW /exec URL and must update index.html again. * * IMPORTANT: whenever you edit this script after the first deployment, * you must create a "New deployment" (or "Manage deployments" → edit → * new version) for changes to take effect on the live /exec URL. * * This single script supports BOTH site versions: * * Version 1 (index_v1_sheet_admin.html) — "Sheet + in-site admin": * - Uses the "KV" tab as a generic key/value data store for everything * (story, announcements, events, inventory, registrations, AND the 3 * group-request forms as Pending/Approved/Rejected records reviewed * inside the site's admin panel). * - Uses "ActivityLog" for every admin action. * * Version 2 (index_v2_direct.html) — "Direct submission": * - Still uses "KV" + "ActivityLog" for story/announcements/events/ * inventory/registrations/admin password — same as V1. * - BUT the 3 group-request forms (Owners' WhatsApp, Volunteer sign-up, * Tenants & spouses) bypass KV entirely and are appended as plain rows * directly into their own dedicated tabs: "OwnersRequests", * "VolunteerSignups", "TenantRequests". Admins review and mark * Approved/Rejected by editing the Status column directly in the Sheet * — there is no in-site approval screen for these three in V2. * * Data model: * - Tab "KV": columns [Key, Value] — generic key/value store. * - Tab "ActivityLog": columns [Timestamp, Volunteer, Action, Details]. * - Tab "OwnersRequests": columns [Timestamp, Name, Tower/Flat, Phone, Status] * - Tab "VolunteerSignups": columns [Timestamp, Name, Tower/Flat, Phone, Interest, Status] * - Tab "TenantRequests": columns [Timestamp, Name, Tower/Flat, Relation, Phone, Status] * * All tabs are created automatically on first use if they don't exist. */ const KV_SHEET_NAME = 'KV'; const LOG_SHEET_NAME = 'ActivityLog'; const DIRECT_SHEET_HEADERS = { OwnersRequests: ['Timestamp', 'Name', 'Tower/Flat', 'Phone', 'Status'], VolunteerSignups: ['Timestamp', 'Name', 'Tower/Flat', 'Phone', 'Interest', 'Status'], TenantRequests: ['Timestamp', 'Name', 'Tower/Flat', 'Relation', 'Phone', 'Status'] }; function getOrCreateSheet(name, headerRow){ const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(name); if (!sheet){ sheet = ss.insertSheet(name); sheet.appendRow(headerRow); sheet.setFrozenRows(1); } return sheet; } function getKvSheet(){ return getOrCreateSheet(KV_SHEET_NAME, ['Key', 'Value']); } function getLogSheet(){ return getOrCreateSheet(LOG_SHEET_NAME, ['Timestamp', 'Volunteer', 'Action', 'Details']); } function getDirectSheet(name){ const headers = DIRECT_SHEET_HEADERS[name]; if (!headers) throw new Error('Unknown direct-submission sheet: ' + name); return getOrCreateSheet(name, headers); } function findKeyRow(sheet, key){ const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++){ // skip header if (data[i][0] === key) return i + 1; // 1-indexed row number } return -1; } function jsonOut(obj){ return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } function doGet(e){ const action = e.parameter.action; const sheet = getKvSheet(); if (action === 'get'){ const key = e.parameter.key; const row = findKeyRow(sheet, key); if (row === -1) return jsonOut({ value: null }); const value = sheet.getRange(row, 2).getValue(); return jsonOut({ value: value === '' ? null : String(value) }); } if (action === 'list'){ const prefix = e.parameter.prefix || ''; const data = sheet.getDataRange().getValues(); const keys = []; for (let i = 1; i < data.length; i++){ const k = data[i][0]; if (typeof k === 'string' && k.indexOf(prefix) === 0) keys.push(k); } return jsonOut({ keys }); } return jsonOut({ error: 'Unknown or missing action' }); } function doPost(e){ const body = JSON.parse(e.postData.contents); if (body.type === 'log'){ const logSheet = getLogSheet(); logSheet.appendRow([ body.timestamp || new Date().toISOString(), body.volunteer || 'Unknown volunteer', body.action || '', body.details || '' ]); return jsonOut({ ok: true }); } if (body.type === 'kv'){ const sheet = getKvSheet(); const key = body.key; if (body.action === 'set'){ const row = findKeyRow(sheet, key); if (row === -1){ sheet.appendRow([key, body.value]); } else { sheet.getRange(row, 2).setValue(body.value); } return jsonOut({ ok: true }); } if (body.action === 'delete'){ const row = findKeyRow(sheet, key); if (row !== -1) sheet.deleteRow(row); return jsonOut({ ok: true }); } } // Version 2 only: direct submission straight into a dedicated tab as a plain row. if (body.type === 'directSubmit'){ const sheet = getDirectSheet(body.sheet); sheet.appendRow(body.row); return jsonOut({ ok: true }); } return jsonOut({ error: 'Unknown or missing type/action' }); }
